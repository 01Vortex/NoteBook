# 汇编语言基础

> 汇编语言是逆向工程的基础，理解汇编对于分析二进制程序至关重要

## x86/x64 架构

### 寄存器

```assembly
; 通用寄存器 (32位)
EAX - 累加器
EBX - 基址寄存器
ECX - 计数器
EDX - 数据寄存器
ESI - 源索引
EDI - 目的索引
EBP - 基址指针
ESP - 栈指针

; 64位扩展
RAX, RBX, RCX, RDX, RSI, RDI, RBP, RSP
R8-R15 (额外寄存器)

; 段寄存器
CS - 代码段
DS - 数据段
SS - 栈段
ES, FS, GS - 附加段

; 标志寄存器
ZF - 零标志
CF - 进位标志
SF - 符号标志
OF - 溢出标志
```

### 常用指令

```assembly
; 数据传送
MOV dest, src       ; 移动数据
LEA dest, src       ; 加载有效地址
PUSH src            ; 压栈
POP dest            ; 出栈

; 算术运算
ADD dest, src       ; 加法
SUB dest, src       ; 减法
MUL src             ; 无符号乘法
IMUL src            ; 有符号乘法
DIV src             ; 无符号除法
IDIV src            ; 有符号除法
INC dest            ; 自增
DEC dest            ; 自减

; 逻辑运算
AND dest, src       ; 与
OR dest, src        ; 或
XOR dest, src       ; 异或
NOT dest            ; 非
SHL dest, count     ; 左移
SHR dest, count     ; 右移

; 控制流
JMP addr            ; 无条件跳转
JE/JZ addr          ; 相等/零跳转
JNE/JNZ addr        ; 不等/非零跳转
JG/JL addr          ; 大于/小于跳转
CALL addr           ; 调用函数
RET                 ; 返回

; 比较
CMP op1, op2        ; 比较
TEST op1, op2       ; 测试
```

## ARM 架构

### 寄存器

```assembly
; 通用寄存器
R0-R12  - 通用寄存器
R13(SP) - 栈指针
R14(LR) - 链接寄存器
R15(PC) - 程序计数器

; 状态寄存器
CPSR - 当前程序状态寄存器
```

### 常用指令

```assembly
; 数据处理
MOV Rd, Op2         ; 移动
ADD Rd, Rn, Op2     ; 加法
SUB Rd, Rn, Op2     ; 减法
MUL Rd, Rm, Rs      ; 乘法

; 内存访问
LDR Rd, [Rn]        ; 加载
STR Rd, [Rn]        ; 存储

; 分支
B label             ; 跳转
BL label            ; 带链接跳转
BX Rm               ; 寄存器跳转
```

## 调用约定

### x86 调用约定

```assembly
; cdecl (C 调用约定)
; - 参数从右到左压栈
; - 调用者清理栈
push arg3
push arg2
push arg1
call function
add esp, 12         ; 清理栈

; stdcall (标准调用)
; - 参数从右到左压栈
; - 被调用者清理栈
push arg2
push arg1
call function       ; 函数内部清理栈

; fastcall
; - 前两个参数用 ECX, EDX
; - 其余参数压栈
mov ecx, arg1
mov edx, arg2
push arg3
call function
```

### x64 调用约定

```assembly
; Windows x64
; RCX, RDX, R8, R9 传递前4个参数
; 其余参数压栈
mov rcx, arg1
mov rdx, arg2
mov r8, arg3
mov r9, arg4
call function

; Linux x64 (System V)
; RDI, RSI, RDX, RCX, R8, R9 传递前6个参数
mov rdi, arg1
mov rsi, arg2
mov rdx, arg3
call function
```

## 栈帧结构

```assembly
; 函数序言
push ebp            ; 保存旧的基址指针
mov ebp, esp        ; 设置新的基址指针
sub esp, 0x20       ; 分配局部变量空间

; 函数体
mov [ebp-4], eax    ; 访问局部变量
mov eax, [ebp+8]    ; 访问参数

; 函数尾声
mov esp, ebp        ; 恢复栈指针
pop ebp             ; 恢复基址指针
ret                 ; 返回
```

## 常见模式识别

### 循环结构

```assembly
; for 循环
mov ecx, 0          ; i = 0
loop_start:
cmp ecx, 10         ; i < 10
jge loop_end
; 循环体
inc ecx             ; i++
jmp loop_start
loop_end:

; while 循环
loop_start:
cmp eax, 0
je loop_end
; 循环体
jmp loop_start
loop_end:
```

### 条件判断

```assembly
; if-else
cmp eax, 0
je else_branch
; if 分支
jmp end_if
else_branch:
; else 分支
end_if:

; switch-case
cmp eax, 1
je case_1
cmp eax, 2
je case_2
jmp default_case
```

## 实战技巧

1. **识别函数边界**: 查找 PUSH EBP / MOV EBP, ESP 序言
2. **追踪参数**: 观察栈操作和寄存器使用
3. **理解返回值**: 通常在 EAX/RAX 中
4. **识别字符串**: 查找 ASCII 或 Unicode 引用
5. **分析跳转**: 理解程序控制流
